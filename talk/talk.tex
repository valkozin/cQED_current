\section{Motivation: can a cavity make a current?}

\begin{frame}{Photon condensation and the no-go theorem}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Question:} can coupling matter to a cavity vacuum produce a \emph{static} field / current in the ground state?
    \begin{itemize}\small
      \item Dicke ``superradiant'' transition: predicted, then forbidden by the $A^2$ term.
      \item \textbf{No-go} \cite{Andolina 2019; Nataf 2019}: for a \emph{uniform} cavity field, gauge invariance
            (f-sum rule) makes the total orbital response diamagnetic $\Rightarrow$ no condensation.
      \item \textbf{Loophole} \cite{Andolina 2020}: a \emph{non-uniform} field (finite $\mathbf q$, coupling to
            $\nabla\times\mathbf A$) condenses if the orbital response is \alert{paramagnetic} and beats the field stiffness.
    \end{itemize}
    \column{0.5\textwidth}
    \begin{keybox}[title={This talk}]
      The minimal realization of the loophole:
      \begin{itemize}\small
        \item ``non-uniform field'' $\to$ \hl{flux through one loop};
        \item ``paramagnetic response'' $\to$ a \alert{$\pi$ junction};
        \item solve it \emph{exactly} (DMRG on electrons + photon), and
              ask what is quantum and what is classical.
      \end{itemize}
    \end{keybox}
    \vspace{2mm}
    {\small Classical cousin: spontaneous $\Phi_0/2$ currents in $\pi$ rings
    \cite{Bulaevskii--Kuzii--Sobyanin 1977} seen in $d$-wave and SFS loops \cite{Tsuei--Kirtley 2000; Bauer 2004}.}
  \end{columns}
\end{frame}

    \end{tikzpicture}
    \par\small $\theta$ = gauge-invariant Cooper-pair phase difference.
    \column{0.5\textwidth}
    \begin{itemize}\small
      \item \textbf{0 junction:} minimum at $\theta=0$. Curvature $E''>0$: the junction is an \emph{inductance}
            $L_J=(\Phi_0/2\pi)^2/E_J$ $\Rightarrow$ diamagnetic.
      \item \textbf{$\pi$ junction:} minimum at $\theta=\pi$; at $\theta=0$ it sits on a \emph{maximum}:
            a \alert{negative inductance} $\Rightarrow$ paramagnetic.
      \item Realizations: SFS ($\pi$ from exchange field), $d$-wave corners, and
            \alert{quantum dots with a local moment} (this talk).
    \end{itemize}
    \begin{calcbox}[title={Classical $\pi$ rf-SQUID}]
      \small $U(\theta)=\frac{E_L}{2}\theta^2+E_J\cos\theta$, \ $E_L=\frac{(\Phi_0/2\pi)^2}{L}$.\\
      $U''(0)=E_L-E_J<0$ iff $\beta\equiv\frac{E_J}{E_L}=\frac{2\pi LI_c}{\Phi_0}>1$.
    \end{calcbox}
  \end{columns}
\end{frame}

\section{The microscopic model}

\begin{frame}{Step 1: flux enters as a Peierls phase}
  \begin{columns}[T]
    \column{0.52\textwidth}
    Tight binding in a vector potential:
    \[
      c^\dagger_{j+1}c_j\;\longrightarrow\;e^{i\theta_j}\,c^\dagger_{j+1}c_j,\qquad
      \theta_j=\frac{e}{\hbar}\int_j^{j+1}\!\mathbf A\cdot d\mathbf l .
    \]
    Gauge transformation $c_j\to e^{i\chi_j}c_j$ moves phase between bonds; only the
    \alert{loop sum} is physical:
    \[
      \varphi\equiv\sum_j\theta_j=\frac{e}{\hbar}\oint\mathbf A\cdot d\mathbf l=\frac{e}{\hbar}\,\Phi .
    \]
    $\Rightarrow$ put the whole $\varphi$ on \emph{one} bond (next to the junction).
    \column{0.46\textwidth}
    \begin{calcbox}[title={Dictionary (keep this in mind!)}]
      \small
      \begin{tabular}{@{}ll@{}}
        electron phase & $\varphi=e\Phi/\hbar$\\
        Cooper-pair phase & $\theta=2\varphi$\\
        $\Phi=\Phi_0=h/2e$ & $\varphi=\pi$\\
        $\Phi=\Phi_0/2$ & $\varphi=\pi/2$\\
        current & $I=-\tfrac{e}{\hbar}\,\partial_\varphi E$
      \end{tabular}
    \end{calcbox}
    \begin{warnbox}
      \small With a real pairing $\Delta$ the ``one-bond'' gauge is \emph{not} the same as a uniform $\mathbf A$;
      it equals uniform $\mathbf A$ with covariant $\Delta_j\to\Delta e^{-2i\chi_j}$ (checked to $10^{-10}$; backup).
    \end{warnbox}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Step 2: quantize the flux of the LC mode}
  \begin{columns}[T]
    \column{0.47\textwidth}
    \[
      H_{LC}=\frac{Q^2}{2C}+\frac{\Phi^2}{2L}=\hw\left(a^\dagger a+\tfrac12\right)
    \]
    \[
      \Phi=\Phi_{\rm zpf}\,(a+a^\dagger),\qquad \Phi_{\rm zpf}=\sqrt{\hbar Z/2}
    \]
    ($Z=\sqrt{L/C}$). The loop phase becomes an operator:
    \[
      \varphi\;\to\;\pcl+g\,(a+a^\dagger),\qquad g=\frac{e\,\Phi_{\rm zpf}}{\hbar}
    \]
    \begin{calcbox}[title={$g$ in circuit language}]
      \small $g^2=\dfrac{e^2Z}{2\hbar}=\pi\dfrac{e^2}{h}Z\ \Rightarrow\ (2g)^2=\pi\dfrac{Z}{R_Q}$,\\[0.5mm]
      $R_Q=h/4e^2\simeq6.45\,\mathrm{k}\Omega$.\quad
      $Z=50\,\Omega\Rightarrow g\approx0.08$;\ \ $Z\approx120\,\Omega\Rightarrow g\approx0.12$.
    \end{calcbox}
    \column{0.5\textwidth}
    \textbf{The junction hopping becomes a photon displacement:}
    \[
      -t_J\,e^{i\pcl}\,\underbrace{e^{ig(a+a^\dagger)}}_{D(ig)}\,c^\dagger_{1\sigma}c_{0\sigma}+\text{h.c.}
    \]
    \begin{itemize}\small
      \item each tunnelling electron kicks the oscillator (displacement $D$);
      \item expanding: $e^{igX}=1+igX\underbrace{-\tfrac{g^2}{2}X^2}_{A^2\text{ term}}+\dots$\\
            $\Rightarrow$ diamagnetic terms included \alert{to all orders};
      \item manifestly gauge invariant: no $A^2$ ``bookkeeping'' errors possible.
    \end{itemize}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{The full Hamiltonian}
  \vspace{-4mm}
  \begin{align*}
    H={}&\textcolor{blue!70!black}{\hw\,a^\dagger a}
      \;\underbrace{-\,t\sum_{r=1}^{L-2}\sum_\sigma\big(c^\dagger_{r+1\sigma}c_{r\sigma}+\text{h.c.}\big)
      +\Delta\sum_{r=1}^{L-1}\big(c^\dagger_{r\uparrow}c^\dagger_{r\downarrow}+\text{h.c.}\big)}_{s\text{-wave BCS chain (the ``leads'')}}\\
     &+\underbrace{\epsilon_d\,n_0+\textcolor{warm}{U\,n_{0\uparrow}n_{0\downarrow}}}_{\text{Anderson dot, site }0}
      \;-\;t_J\sum_\sigma\big(c^\dagger_{0\sigma}c_{L-1\,\sigma}+\text{h.c.}\big)\\
     &-\,t_J\sum_\sigma\Big(e^{i\pcl}\,\textcolor{blue!70!black}{e^{ig(a+a^\dagger)}}\,c^\dagger_{1\sigma}c_{0\sigma}+\text{h.c.}\Big)
     \;\equiv\;\hw\,a^\dagger a+\Hring\big[\pcl+g(a+a^\dagger)\big]
  \end{align*}
  \vspace{-3mm}
  \begin{columns}[T]
    \column{0.56\textwidth}
    \begin{calcbox}[title={Default parameters}]
      \footnotesize $t=1,\ \Delta=t,\ t_J=0.5t,\ L=14\ (\gg\xi\simeq2t/\Delta)$, $\hw=7.6\times10^{-4}t$,\\
      $\epsilon_d=-U/2$ (p--h symmetric); \alert{no external flux}: $\pcl=10^{-3}$ seed only.
    \end{calcbox}
    \column{0.42\textwidth}
    \footnotesize Auxiliary versions: (i) weak link ($U=0$) biased by $\Phi_0/2$;
    (ii) short ring with a Zeeman-polarized dot (the original mean-field notebook).
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{All the ring physics is in one function: $E(\varphi)$}
  \begin{columns}[T]
    \column{0.52\textwidth}
    Ground-state energy of $\Hring$ at fixed (classical) phase:
    \[
      E(\varphi)=\sum_{m\ge0}E_m\cos m\varphi
    \]
    \begin{itemize}\small
      \item \textbf{Only cosines:} $\Hring[\varphi]^*=\Hring[-\varphi]$ (time reversal) $\Rightarrow E(\varphi)=E(-\varphi)$.
      \item \textbf{$m=2,4,\dots$}: charge $2e$ crossing $\Rightarrow$ $h/2e$ periodicity (Josephson).
      \item \textbf{$m=1$}: a single electron encircling the ring \cite{Loss--Goldbart 1991}, $\propto e^{-L/\xi}$:
            a mesoscopic nuisance.
      \item Define $E_J\equiv|E''(\varphi_0)|/4$ ($=|E_2|$ up to $4E_4$), \ $I_c=2eE_J/\hbar$.
    \end{itemize}
    \column{0.46\textwidth}
    \begin{keybox}[title={Sign convention}]
      \small
      \centering
      \begin{tabular}{@{}lcc@{}}
        & $E_2$ & $E''(0)$ \\ \midrule
        0 junction & $<0$ & $>0$ (diamagnetic)\\
        $\pi$ junction & $>0$ & $<0$ (\alert{paramagnetic})
      \end{tabular}
    \end{keybox}
    \vspace{1mm}
    \small For a pure $\pi$ junction, $E(\varphi)=E_J\cos2\varphi$:
    \[
      I(\varphi)=-\frac{e}{\hbar}E'(\varphi)=\underbrace{\frac{2eE_J}{\hbar}}_{I_c}\sin2\varphi .
    \]
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Two routes to a $\pi$ loop --- and why they differ}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Route A: 0 junction + external $\Phi_0/2$} ($\pcl=\pi/2$):
    \begin{align*}
      E_2\cos2(\varphi+\tfrac\pi2)&=-E_2\cos2\varphi\quad\checkmark\\
      E_1\cos(\varphi+\tfrac\pi2)&=\alert{-E_1\sin\varphi}\quad\text{(odd!)}
    \end{align*}
    The $h/e$ harmonic becomes a \alert{tilt} of the double well: it \emph{selects} the current direction.
    Also needs a flux stable to $|\delta\Phi|\ll\delta/2I^\star$.
    \par\centering
    \begin{tikzpicture}[yscale=0.5,xscale=0.75]
      \draw[->,ink!60] (-1.9,0) -- (2.0,0) node[right,font=\scriptsize] {$\Phi$};
      \draw[warm,very thick,domain=-1.7:1.7,samples=80] plot (\x,{0.9*\x*\x*\x*\x/2-1.6*\x*\x/2*1.1+0.35*\x+0.6});
      \node[font=\scriptsize,warm] at (0,-0.45) {tilted: one well wins};
    \end{tikzpicture}
    \column{0.5\textwidth}
    \textbf{Route B: intrinsic $\pi$ junction} ($\pcl=0$):
    \begin{itemize}\small
      \item $\Hring[0]$ is real $\Rightarrow$ $E(\varphi)=E(-\varphi)$ \emph{exactly}.
      \item The two current directions are related by time reversal: a genuine $\mathbb Z_2$ symmetry,
            spontaneously broken (or tunnelling-restored).
      \item Our choice: an interacting dot, $U>U_c$.
    \end{itemize}
    \begin{keybox}
      \small Up to odd harmonics the two routes give the \emph{same} $E(\varphi)$ (shown later, DMRG).
    \end{keybox}
    \centering
    \begin{tikzpicture}[yscale=0.5,xscale=0.75]
      \draw[->,ink!60] (-1.9,0) -- (2.0,0) node[right,font=\scriptsize] {$\Phi$};
      \draw[accent,very thick,domain=-1.7:1.7,samples=80] plot (\x,{0.9*\x*\x*\x*\x/2-1.6*\x*\x/2*1.1+0.6});
      \node[font=\scriptsize,accent] at (0,-0.45) {symmetric: $\mathbb Z_2$ intact};
    \end{tikzpicture}
  \end{columns}
\end{frame}

\section{Theory: adiabatic reduction and Landau theory}

\begin{frame}{Born--Oppenheimer: electrons are fast, the flux is slow}
  \begin{columns}[T]
    \column{0.55\textwidth}
    Scales: $\hw\sim10^{-3}t\ \ll\ E_A$ (Andreev gap $\sim0.1$--$1\,t$).\\[1mm]
    Introduce flux quadratures:
    \[
      X=a+a^\dagger,\quad P=i(a^\dagger-a),\quad [X,P]=2i
    \]
    \[
      \hw\,a^\dagger a=\tfrac{\hw}{4}\left(X^2+P^2\right)-\tfrac{\hw}{2}.
    \]
    Ansatz $|\Psi\rangle=\int\!dX\,\chi(X)\,|X\rangle\otimes|\psi_0(\pcl+gX)\rangle$ with
    $\Hring[\varphi]|\psi_0(\varphi)\rangle=E(\varphi)|\psi_0(\varphi)\rangle$:
    \begin{keybox}
      \centering$\displaystyle H_{\rm BO}=\frac{\hw}{4}\left(P^2+X^2\right)+E(\pcl+gX)$
    \end{keybox}
    \column{0.43\textwidth}
    \begin{itemize}\small
      \item A 1D Schr\"odinger problem for the flux; $P=-2i\,\partial_X$.
      \item The photon is treated \alert{exactly} (all Fock states, all correlations); only adiabaticity is assumed.
      \item Neglected: diagonal BO term $\propto\hw\,g^2\langle\partial_\varphi\psi_0|\partial_\varphi\psi_0\rangle$,
            relative size $\sim\hw/E_A$.
      \item Uses the \emph{full} $E(\varphi)$, not only the first harmonic.
      \item Checked against full DMRG: $\lesssim$ few \% everywhere.
    \end{itemize}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{It is an rf-SQUID (fluxonium) whose inductance is the mode}
  Change variables to the Cooper-pair phase and its conjugate charge:
  \[
    \theta=2gX,\qquad n=\frac{P}{4g},\qquad [\theta,n]=i .
  \]
  \vspace{-4mm}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{calcbox}[title={Two lines of algebra}]
      \small
      $\dfrac{\hw}{4}X^2=\dfrac{\hw}{16g^2}\theta^2\equiv\dfrac12E_L\theta^2,\quad E_L=\dfrac{\hw}{8g^2}$\\[1mm]
      $\dfrac{\hw}{4}P^2=4\hw g^2\,n^2\equiv4E_Cn^2,\quad E_C=\hw\,g^2$\\[1mm]
      Consistency: $\sqrt{8E_CE_L}=\hw$\ \checkmark
    \end{calcbox}
    \begin{keybox}
      \centering$\displaystyle H_{\rm BO}=4E_C\,n^2+\tfrac12E_L\,\theta^2+E(\theta/2)$
    \end{keybox}
    \column{0.43\textwidth}
    \begin{itemize}\small
      \item Pure $\pi$ junction: $E(\theta/2)=+E_J\cos\theta$ $\Rightarrow$ a \alert{$\pi$-fluxonium}.
      \item $g$ controls the \emph{ratio} $E_L/E_C=1/(8g^4)$: small $g$ = heavy, classical flux.
      \item Two dimensionless knobs:
        \[\beta=\frac{E_J}{E_L},\qquad\frac{E_J}{\hw}.\]
    \end{itemize}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Landau theory: when does the mode condense?}
  \begin{columns}[T]
    \column{0.5\textwidth}
    Classical energy (drop $P$): \
    $\mathcal F(X)=\dfrac{\hw}{4}X^2+E(gX)$
    \[
      \mathcal F''(0)=\frac{\hw}{2}+g^2E''(0)=\frac{\hw}{2}\,(1-\beta)
    \]
    \begin{keybox}[title={Instability criterion}]
      \centering$\displaystyle \beta\equiv\frac{2g^2|E''(0)|}{\hw}>1\quad\text{with}\quad E''(0)<0$
    \end{keybox}
    \small Pure $\pi$ junction, $E''(0)=-4E_J$:
    \[
      \beta=\frac{8g^2E_J}{\hw}=\frac{E_J}{E_L}=\frac{2\pi LI_c}{\Phi_0}
    \]
    = the textbook screening parameter of the loop.
    \column{0.5\textwidth}
    \begin{itemize}\small
      \item Needs \alert{paramagnetic} response ($E''<0$): a 0 junction only \emph{stiffens} the mode.
      \item This is the single-loop version of the photon-condensation criterion of Andolina \emph{et al.}
      \item Why the no-go theorem does not apply: the mode couples to the \emph{flux} (curl of $\mathbf A$), and
            the $\pi$ shift supplies the paramagnetism.
    \end{itemize}
    \begin{calcbox}[title={Numbers ($U=2t$ dot)}]
      \footnotesize $E_J=|E''(0)|/4=7.61\times10^{-3}t$ ($E_2=7.39\times10^{-3}t$), $\hw=E_J/10$\\
      $\Rightarrow g_c^{\rm cl}=\sqrt{\hw/8E_J}=0.112$. Full DMRG: $g=0.117$ linear, $0.129$ ordered \checkmark
    \end{calcbox}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{The classical double well and the pitchfork}
  \centering
  \includegraphics[width=0.92\textwidth]{landau.pdf}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \small $\mathcal F'(X)=0$ with $\theta=2gX$:\quad
    $\dfrac{\hw}{2g}\,\varphi=2gE_J\sin2\varphi\ \Rightarrow\ \boxed{\theta^\star=\beta\sin\theta^\star}$
    \column{0.5\textwidth}
    \small $I^\star=I_c\sin\theta^\star$. Large $\beta$: $\theta^\star\to\pi$ (flux $\to\Phi_0/2$) and
    $I^\star\simeq\pi I_c/\beta\propto\hw/g^2$ \emph{decreases}.
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{An exact identity: the current \emph{is} the displacement of the mode}
  \begin{columns}[T]
    \column{0.55\textwidth}
    In any eigenstate $\langle[H,P]\rangle=0$. Compute the commutator:
    \begin{align*}
      [a^\dagger a,P]&=i\,(a^\dagger+a)=iX,\\
      [\Hring(\pcl+gX),P]&=2i\,g\,\partial_\varphi\Hring .
    \end{align*}
    With $I\equiv-\frac{e}{\hbar}\partial_\varphi\Hring$:
    \begin{keybox}
      \centering$\displaystyle \langle I\rangle=\frac{e}{\hbar}\,\frac{\hw\,\langle X\rangle}{2g}
        \qquad\Longleftrightarrow\qquad \langle I\rangle=\frac{\langle\Phi\rangle}{L}$
    \end{keybox}
    \small (use $2\Phi_{\rm zpf}^2=\hw L$). Kirchhoff's law, but exact for the full quantum problem.
    \column{0.43\textwidth}
    \textbf{Why it matters}
    \begin{itemize}\small
      \item Current flows $\Leftrightarrow$ the field has a static part, $\langle a\rangle\neq0$: a \alert{photon condensate}.
      \item Condensate size: $\langle a^\dagger a\rangle\ge\langle X\rangle^2/4=(gI\hbar/e\hw)^2$.\\
            $U=2t$, $g=0.137$, $I^\star=0.0138$: $X^\star=4.97$, $\ge6.2$ photons (DMRG: $X^\star=4.97$, $\langle n\rangle=6.2$).
      \item Numerically: $\langle X\rangle$ converges much better in DMRG than the bond current
            (a small difference of large terms) $\Rightarrow$ we measure $I$ through $X$.
      \item Verified in ED to $10^{-7}$.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Below threshold: the soft mode}
  \begin{columns}[c]
    \column{0.5\textwidth}
    Expand to second order around $X=0$ ($\beta<1$):
    \[
      H_{\rm BO}\simeq\frac{\hw}{4}\Big(P^2+(1-\beta)X^2\Big)
    \]
    \begin{itemize}\small
      \item Excitation gap: $\boxed{\hw\sqrt{1-\beta}}$ \ (mode softening).
      \item Flux fluctuations grow: $\langle X^2\rangle=(1-\beta)^{-1/2}$.
      \item Photons in the ground state:
        $\langle n\rangle=\tfrac14\big[(1-\beta)^{-\frac12}+(1-\beta)^{\frac12}\big]-\tfrac12$.
    \end{itemize}
    \begin{calcbox}[title={Check against full DMRG}]
      \footnotesize $U=2t$, $\beta=0.5$: gap $\sqrt{0.5}=0.707$ vs DMRG $0.714$; $\langle n\rangle=0.030$ vs DMRG $0.029$.
      ($\beta=1$: DMRG $0.28$; quartic regime.)
    \end{calcbox}
    \column{0.5\textwidth}
    \centering
    \includegraphics[width=0.93\textwidth]{soft_mode.pdf}
    \par\small At $\beta\to1$ the quadratic theory fails (quartic oscillator); above it, the gap
    becomes the \alert{tunnel splitting} and collapses exponentially.
  \end{columns}
\end{frame}

\begin{frame}{How small is $\delta$? An instanton estimate}
  \begin{columns}[T]
    \column{0.55\textwidth}
    Particle of mass $M=\hbar^2/8E_C$ in $U(\theta)=E_J\,u(\theta)$,
    $u=\frac{\theta^2}{2\beta}+\cos\theta$. WKB:
    \[
      \delta\sim\hw\,e^{-S},\quad
      S=\int_{-\theta^\star}^{\theta^\star}\!d\theta\,\sqrt{\frac{U-U_{\min}}{4E_C}}
    \]
    \begin{calcbox}[title={Scaling}]
      \small $\dfrac{E_J}{4E_C}=\dfrac{E_J}{4g^2\hw}=\dfrac{2}{\beta}\Big(\dfrac{E_J}{\hw}\Big)^2$
      $\ \Rightarrow\ \boxed{S=\dfrac{E_J}{\hw}\,s(\beta)}$\\[1mm]
      $s(\beta)=\sqrt{2/\beta}\int\!d\theta\,\sqrt{u-u_{\min}}$.\\
      Near threshold ($\epsilon=1-1/\beta$): $u\simeq1-\frac{\epsilon}{2}\theta^2+\frac{\theta^4}{24}$,
      minima at $\theta^2=6\epsilon$, depth $\frac32\epsilon^2$ $\Rightarrow s\propto(\beta-1)^{3/2}$.
    \end{calcbox}
    \column{0.43\textwidth}
    \begin{keybox}[title={Sharp transition only in a scaling limit}]
      \small $E_J/\hw\to\infty$ at fixed $\beta$:
      \[
        \frac{E_C}{E_J}=\frac{\beta}{8}\Big(\frac{\hw}{E_J}\Big)^2\to0 ,
      \]
      flux becomes classical, $\delta\to0$, pitchfork with mean-field exponents.
    \end{keybox}
    \small Cf.\ quantum Rabi model: QPT only for $\Omega/\omega\to\infty$ \cite{Hwang--Puebla--Plenio 2015}.
    Finite $E_J/\hw$: a crossover; $\delta$ = macroscopic quantum tunnelling of the current.
  \end{columns}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{What the cavity does and does not do}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Static threshold is set by the loop inductance.}\\[1mm]
    \small Loop with its own $L_r$, mutual $M\le\sqrt{L_rL_c}$ to the resonator: minimizing the static energy
    over the resonator flux leaves $\Phi_r^2/2L_r$.
    \begin{itemize}\small
      \item A resonator cannot push the static threshold below what the loop's own inductance gives.
      \item Our model is the galvanic limit $k=M/\sqrt{L_rL_c}\to1$ (resonator inductance \emph{is} the loop inductance, as in fluxonium).
    \end{itemize}
    \column{0.5\textwidth}
    \begin{warnbox}[title={Honest framing}]
      \small
      \begin{itemize}
        \item The instability is \emph{classical}: a static field, $\beta>1$.
        \item Vacuum fluctuations do \alert{not} drive it, they oppose it (tunnelling $\delta$).
        \item The mode supplies the \emph{quantum dynamics} of the flux ($E_C$), the condensate's photons, and a readout channel.
      \end{itemize}
    \end{warnbox}
  \end{columns}
\end{frame}

\begin{frame}{Three ways to solve $H=\hw a^\dagger a+\Hring[\pcl+gX]$}

  \end{keybox}
\end{frame}

\begin{frame}{Mean field and its two hidden assumptions}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \small Product state $\Rightarrow$ two coupled problems:
    \begin{align*}
      \text{el.:}&\ \Hring\ \text{with}\ t_J\to t_J\,\alpha,\quad \alpha=\langle\chi|e^{igX}|\chi\rangle\\
      \text{ph.:}&\ \hw a^\dagger a-\big(\eta\,e^{igX}+\text{h.c.}\big),\\
      &\ \eta=t_Je^{i\pcl}\textstyle\sum_\sigma\langle c^\dagger_{1\sigma}c_{0\sigma}\rangle\\
      E_{\rm MF}&=E_{\rm el}+E_{\rm ph}+2\,\mathrm{Re}(\eta\alpha)\ \ \text{(double counting)}
    \end{align*}
    For a coherent state $|X_0\rangle$: \ $\alpha=e^{igX_0}\,\alert{e^{-g^2/2}}$ (Debye--Waller).
    \column{0.46\textwidth}
    \begin{warnbox}[title={(i) Not gauge invariant}]
      \small All phase on one bond: that bond is shrunk by $e^{-g^2/2}$ and loses \emph{phase-independent} kinetic
      energy. Spread over $N$ bonds: loss $\sim N\cdot g^2/2N^2\to0$. The exact state recovers it by following the flux
      adiabatically. \cite{Li \emph{et al.} 2020; Dmytruk--Schir\`o 2021}
    \end{warnbox}
    \begin{warnbox}[title={(ii) Self-trapping}]
      \small A product state must pick one well: it breaks the $\mathbb Z_2$ by construction, even when the
      barrier is below the zero-point energy.
    \end{warnbox}
  \end{columns}
\end{frame}

\begin{frame}{Why a local moment gives a $\pi$ junction}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \small Singly occupied dot ($U\gg t_J,\Delta$): a Cooper pair crosses by \emph{fourth-order cotunnelling},
    one electron at a time, through virtual states.
    \begin{itemize}\small
      \item The dot's own electron must leave and be replaced; the pair arrives with its two spins in \alert{reversed order}.
      \item Singlet: $c^\dagger_{\downarrow}c^\dagger_{\uparrow}=-c^\dagger_{\uparrow}c^\dagger_{\downarrow}$
            $\Rightarrow$ the pair-transfer amplitude changes sign: $E_J\to-E_J$.
      \item \cite{Glazman--Matveev 1989; Spivak--Kivelson 1991}; seen in nanowire and CNT dots \cite{van Dam 2006; Cleuziou 2006}.
    \end{itemize}
    \column{0.46\textwidth}

\section{Summary}

\begin{frame}{Take-home messages}
  \begin{enumerate}
    \item A $\pi$ junction in a loop is the \alert{minimal photon condensate}: paramagnetic orbital response +
          flux coupling evade the no-go theorem. Criterion: $\beta=2g^2|E''(0)|/\hw>1$.
    \item Everything reduces to a \hl{$\pi$-fluxonium} $4E_Cn^2+\frac12E_L\theta^2+E(\theta/2)$ with
          $E_C=g^2\hw$, $E_L=\hw/8g^2$; exact identity $\langle I\rangle=\langle\Phi\rangle/L$.
    \item The transition is classical in origin; quantum fluctuations restore symmetry via a
          \alert{cat state} with $\delta\sim\hw\,e^{-(E_J/\hw)s(\beta)}$. Sharp only for $E_J/\hw\to\infty$.
    \item An \emph{interacting} dot ($U>U_c\simeq0.9t$) gives a TRS-protected $\pi$ loop: current direction
          chosen spontaneously, no flux bias to stabilize; fourfold spin$\otimes$current multiplet.
    \item Full electron--photon DMRG confirms it at $E_J/\hw=10$ ($U=2t$ dot, weak link) and shows that MF \alert{invents}
          the effect when $E_J\sim\hw$. The $(U,g)$ and $\delta$ maps are adiabatic theory built on DMRG $E(\varphi;U)$,
          the reliable cheap tool.
  \end{enumerate}
\end{frame}

% ------------------------------------------------------------------
\begin{frame}{Open questions}
  \begin{itemize}
    \item \textbf{Kondo regime} ($T_K\gg\Delta$): the junction returns to 0 --- how does the condensate disappear?
    \item \textbf{Self-consistent leads}: with a rigid order parameter all phase drops on the junction; a
          self-consistent BdG ring puts most of it in the leads (backup). How does lead kinetic inductance renormalize $\beta$?
    \item \textbf{Spin $\otimes$ current multiplet}: can the $\{\uparrow,\downarrow\}\otimes\{\cw,\ccw\}$ quartet be used/read out?
    \item \textbf{Dissipation and readout}: what does a real resonator measure --- $\langle0|\Phi|1\rangle$, dispersive shift?
    \item \textbf{Beyond one mode}: multimode / transmission-line resonators, finite $k<1$.
  \end{itemize}
  \vfill
  \centering\Large Thank you!
\end{frame}
